import csv


def classificar_evento(tentativas):
    """Classifica o evento conforme a quantidade de tentativas."""
    if tentativas >= 10:
        return "SUSPEITO"
    elif tentativas >= 5:
        return "ATENCAO"
    return "NORMAL"


def analisar_arquivo(caminho):
    total = 0
    alertas = 0
    invalidos = 0

    try:
        with open(caminho, "r", encoding="utf-8", newline="") as arquivo:
            leitor = csv.DictReader(arquivo)

            colunas = {"data_hora", "usuario", "ip", "tentativas"}

            if not leitor.fieldnames or not colunas.issubset(leitor.fieldnames):
                print("Erro: cabecalho CSV invalido.")
                return

            print("=== ANALISE DE LOGS ===\n")

            for linha in leitor:
                try:
                    usuario = linha["usuario"].strip()
                    ip = linha["ip"].strip()
                    data_hora = linha["data_hora"].strip()
                    tentativas = int(linha["tentativas"])

                    if not usuario or not ip or not data_hora:
                        raise ValueError("Campo obrigatorio vazio")

                    if tentativas < 0:
                        raise ValueError("Tentativas negativas")

                except (ValueError, AttributeError, TypeError):
                    invalidos += 1
                    print("Registro invalido ignorado.")
                    continue

                classificacao = classificar_evento(tentativas)
                total += 1

                if classificacao != "NORMAL":
                    alertas += 1

                print(f"Data/hora: {data_hora}")
                print(f"Usuario: {usuario}")
                print(f"IP: {ip}")
                print(f"Tentativas: {tentativas}")
                print(f"Classificacao: {classificacao}")
                print("-" * 35)

    except FileNotFoundError:
        print(f"Erro: arquivo nao encontrado: {caminho}")
        return
    except PermissionError:
        print("Erro: sem permissao para ler o arquivo.")
        return
    except OSError as erro:
        print(f"Erro ao acessar o arquivo: {erro}")
        return

    # Exibição do relatório no terminal
    print("\n=== RELATORIO FINAL ===")
    print(f"Eventos validos: {total}")
    print(f"Eventos para revisao: {alertas}")
    print(f"Registros invalidos: {invalidos}")
    print(f"Eventos normais: {total - alertas}")

    # Desafio Extra: Gravação do relatório em relatorio.txt
    try:
        with open("relatorio.txt", "w", encoding="utf-8") as relatorio:
            relatorio.write("=== RELATORIO DE ANALISE DE LOGS ===\n")
            relatorio.write(f"Eventos validos: {total}\n")
            relatorio.write(f"Eventos para revisao: {alertas}\n")
            relatorio.write(f"Registros invalidos: {invalidos}\n")
            relatorio.write(f"Eventos normais: {total - alertas}\n")
        print("\nRelatorio salvo em 'relatorio.txt' com sucesso.")
    except OSError as erro:
        print(f"Erro ao salvar arquivo de relatorio: {erro}")


if __name__ == "__main__":
    analisar_arquivo("eventos.csv")

[ AVALIAÇÃO DO DIA 04 - RESPOSTAS TÉCNICAS ]

1. Qual é a vantagem de usar uma função para classificar eventos?
   - Desacopla a regra de classificação da leitura do arquivo. Isso garante reutilização do código em múltiplos módulos e facilita manutenções futuras (ex: alterar os thresholds em um único local sem afetar a lógica I/O).

2. Por que csv.DictReader é útil para analisar logs?
   - Mapeia automaticamente cada linha do CSV em um dicionário utilizando os nomes das colunas como chaves (`linha["usuario"]`). Isso evita dependência de índices numéricos fixos, tornando o script imune a alterações na ordem das colunas do arquivo fonte.

3. Qual é a diferença entre ValueError e FileNotFoundError?
   - `FileNotFoundError`: Exceção de sistema/arquivo disparada quando o caminho especificado não existe no sistema de arquivos.
   - `ValueError`: Exceção de conversão/validação de dados disparada quando o tipo recebido está correto, mas o conteúdo é inválido (ex: tentar converter `"abc"` para `int()`).

4. Por que um registro inválido não deve interromper toda a análise?
   - Em operações de SOC e análise defensiva de logs, arquivos massivos frequentemente possuem registros corrompidos ou incompletos. Interromper o script impediria o processamento do restante do arquivo, gerando pontos cegos na investigação.

5. Como você adaptaria o programa para produzir um relatório útil em uma investigação real?
   - Adicionando correlação temporal (janelas curtas com muitas falhas do mesmo usuário/IP), exportação estruturada em formato JSON/CEF para envio a um SIEM, e integração com bancos de inteligência de ameaças (Threat Intelligence) para reputação de IP.

--- File: port_check.py ---
port = 22

print("=== Security Port Check ===")
if port == 22:
    print("Porta 22: SSH")
    print("Verifique se o acesso remoto esta protegido.")
elif port == 3389:
    print("Porta 3389: RDP")
    print("Verifique as restricoes de acesso.")
elif port == 445:
    print("Porta 445: SMB")
    print("Verifique as regras de acesso ao compartilhamento.")
else:
    print(f"Porta {port}: servico nao identificado neste exemplo.")


--- File: port_inventory.py ---
ports = [22, 80, 443, 445, 3389, 8080]

print("=== Port Inventory ===")
for port in ports:
    if port == 22:
        service = "SSH"
    elif port == 80:
        service = "HTTP"
    elif port == 443:
        service = "HTTPS"
    elif port == 445:
        service = "SMB"
    elif port == 3389:
        service = "RDP"
    else:
        service = "Other / Unknown"
    print(f"Porta: {port} | Servico: {service}")


--- File: security_report.py ---
hostname = "debian-cyberlab"

import csv
from datetime import datetime


def classificar_evento(tentativas):
    """Classifica o risco do evento com base no número de tentativas falhas."""
    if tentativas >= 10:
        return "SUSPEITO"
    elif tentativas >= 5:
        return "ATENCAO"
    return "NORMAL"


def analisar_eventos(caminho_csv):
    """Lê o arquivo CSV, valida os campos e agrega as estatísticas dos eventos."""
    eventos_processados = []
    estatisticas = {
        "validos": 0,
        "invalidos": 0,
        "normal": 0,
        "atencao": 0,
        "suspeito": 0,
    }

    try:
        with open(caminho_csv, "r", encoding="utf-8", newline="") as arquivo:
            leitor = csv.DictReader(arquivo)
            colunas_esperadas = {"data_hora", "usuario", "ip", "tentativas"}

            if not leitor.fieldnames or not colunas_esperadas.issubset(leitor.fieldnames):
                raise ValueError("Cabeçalho do arquivo CSV é inválido ou incompleto.")

            for numero_linha, linha in enumerate(leitor, start=2):
                try:
                    usuario = linha["usuario"].strip()
                    ip = linha["ip"].strip()
                    data_hora = linha["data_hora"].strip()

                    if not usuario or not ip or not data_hora:
                        raise ValueError("Campo obrigatório vazio.")

                    tentativas = int(linha["tentativas"])
                    if tentativas < 0:
                        raise ValueError("Quantidade de tentativas negativa.")

                    classificacao = classificar_evento(tentativas)

                    registro = {
                        "data_hora": data_hora,
                        "usuario": usuario,
                        "ip": ip,
                        "tentativas": tentativas,
                        "classificacao": classificacao,
                    }

                    eventos_processados.append(registro)
                    estatisticas["validos"] += 1

                    if classificacao == "SUSPEITO":
                        estatisticas["suspeito"] += 1
                    elif classificacao == "ATENCAO":
                        estatisticas["atencao"] += 1
                    else:
                        estatisticas["normal"] += 1

                except (ValueError, KeyError, AttributeError):
                    estatisticas["invalidos"] += 1
                    continue

    except FileNotFoundError:
        print(f"[ERRO] O arquivo '{caminho_csv}' não foi encontrado.")
        return None, None
    except Exception as e:
        print(f"[ERRO] Falha ao processar o arquivo: {e}")
        return None, None

    return eventos_processados, estatisticas


def salvar_relatorio(caminho_saida, eventos, estatisticas):
    """Gera o arquivo TXT do relatório consolidado com carimbo de data/hora."""
    try:
        agora = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        with open(caminho_saida, "w", encoding="utf-8") as relatorio:
            relatorio.write("==================================================\n")
            relatorio.write("        RELATÓRIO DE ANÁLISE DE SEGURANÇA        \n")
            relatorio.write("==================================================\n")
            relatorio.write(f"Data de Geração: {agora}\n\n")

            relatorio.write("--- REGISTROS PROCESSADOS ---\n")
            for evt in eventos:
                relatorio.write(
                    f"[{evt['data_hora']}] Usuário: {evt['usuario']} | IP: {evt['ip']} | "
                    f"Tentativas: {evt['tentativas']} | Classificação: {evt['classificacao']}\n"
                )

            relatorio.write("\n--- RESUMO CONSOLIDADO ---\n")
            relatorio.write(f"Eventos válidos:      {estatisticas['validos']}\n")
            relatorio.write(f"Registros inválidos:  {estatisticas['invalidos']}\n")
            relatorio.write(f"Normais:              {estatisticas['normal']}\n")
            relatorio.write(f"Atenção:              {estatisticas['atencao']}\n")
            relatorio.write(f"Suspeitos:            {estatisticas['suspeito']}\n")
            relatorio.write(f"Total de Alertas:     {estatisticas['atencao'] + estatisticas['suspeito']}\n")
            relatorio.write("==================================================\n")

        print(f"[SUCESSO] Relatório gerado com sucesso em '{caminho_saida}'.")
    except OSError as erro:
        print(f"[ERRO] Não foi possível salvar o relatório: {erro}")


def main():
    arquivo_entrada = "eventos.csv"
    arquivo_saida = "relatorio.txt"

    print("Iniciando análise de eventos de segurança...")
    eventos, estatisticas = analisar_eventos(arquivo_entrada)

    if eventos is not None and estatisticas is not None:
        salvar_relatorio(arquivo_saida, eventos, estatisticas)


if __name__ == "__main__":
    main()

--------------------------------------------------------------------------------
1. Por que dividimos o programa em várias funções?
   - Facilitar a manutenção, testabilidade e reutilização do código. Cada função cumpre uma responsabilidade única (Princípio da Responsabilidade Única - SRP).

2. Qual é a função de csv.DictReader?
   - Converter cada linha do arquivo CSV em um dicionário Python utilizando o cabeçalho como chave, permitindo acesso seguro aos dados pelo nome do campo em vez de um índice numérico fixo.

3. Como o script trata registros inválidos sem interromper toda a análise?
   - Encapsulando a leitura e conversão de cada linha em um bloco `try-except` interno ao loop. Quando uma falha é detectada (ex: `ValueError`), a linha é contabilizada como inválida e a instrução `continue` avança diretamente para o próximo registro.

4. Qual é a diferença entre exibir o resultado no terminal e salvá-lo em um arquivo?
   - A exibição no terminal é volátil (útil apenas para feedback imediato do operador), enquanto a gravação em arquivo garante rastreabilidade, persistência histórica e capacidade de integração com pipelines automatizadas de auditoria.

5. Como você adaptaria esse programa para ajudar na análise de logs de autenticação do Linux?
   - Substituindo o leitor CSV por um parser de arquivos de texto (ex: expressões regulares com `re` para ler `/var/log/auth.log` ou `secure`), extraindo os padrões do OpenSSH (`Failed password for...`) e agregando as tentativas de falha por IP de origem em janelas temporais.

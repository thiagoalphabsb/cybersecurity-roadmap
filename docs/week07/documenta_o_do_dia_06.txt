import csv
from collections import Counter
from datetime import datetime


def classificar_evento(tentativas):
    """Classifica o nível de risco com base no número de tentativas de login falhas."""
    if tentativas >= 10:
        return "SUSPEITO"
    elif tentativas >= 5:
        return "ATENCAO"
    return "NORMAL"


def analisar_csv(caminho_csv):
    """Lê, valida e processa o arquivo CSV de eventos."""
    eventos = []
    registros_invalidos = 0

    with open(caminho_csv, encoding="utf-8", newline="") as arquivo:
        leitor = csv.DictReader(arquivo)
        colunas_obrigatorias = {"data_hora", "usuario", "ip", "tentativas"}

        if not leitor.fieldnames or not colunas_obrigatorias.issubset(leitor.fieldnames):
            raise ValueError("Cabeçalho CSV inválido ou colunas ausentes.")

        for linha in leitor:
            try:
                usuario = linha["usuario"].strip()
                ip = linha["ip"].strip()
                data_hora = linha["data_hora"].strip()

                if not usuario or not ip or not data_hora:
                    registros_invalidos += 1
                    continue

                tentativas = int(linha["tentativas"])
                if tentativas < 0:
                    registros_invalidos += 1
                    continue

                classificacao = classificar_evento(tentativas)

                eventos.append({
                    "data_hora": data_hora,
                    "usuario": usuario,
                    "ip": ip,
                    "tentativas": tentativas,
                    "classificacao": classificacao
                })

            except (ValueError, TypeError, KeyError):
                registros_invalidos += 1
                continue

    return eventos, registros_invalidos


def gerar_e_salvar_relatorio(eventos, registros_invalidos, caminho_saida="relatorio_final.txt"):
    """Exibe o resumo no terminal e grava o relatório completo em disco."""
    categorias = Counter(evento["classificacao"] for evento in eventos)
    agora = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

    # Impressão no Terminal
    print("=== RESUMO DA ANALISE ===")
    print(f"Total de eventos válidos: {len(eventos)}")
    print(f"Registros inválidos/ignorados: {registros_invalidos}")
    print(f"Normais: {categorias['NORMAL']}")
    print(f"Atenção: {categorias['ATENCAO']}")
    print(f"Suspeitos: {categorias['SUSPEITO']}")

    # Gravação em Arquivo
    try:
        with open(caminho_saida, "w", encoding="utf-8") as relatorio:
            relatorio.write("==================================================\n")
            relatorio.write("   PYTHON SECURITY LOG ANALYZER - FINAL REPORT   \n")
            relatorio.write("==================================================\n")
            relatorio.write(f"Data de Processamento: {agora}\n\n")

            relatorio.write("--- DETALHAMENTO DOS EVENTOS ---\n")
            for evt in eventos:
                relatorio.write(
                    f"[{evt['data_hora']}] User: {evt['usuario']:<10} | IP: {evt['ip']:<15} | "
                    f"Tentativas: {evt['tentativas']:<2} | Risco: {evt['classificacao']}\n"
                )

            relatorio.write("\n--- RESUMO CONSOLIDADO ---\n")
            relatorio.write(f"Total de eventos válidos: {len(eventos)}\n")
            relatorio.write(f"Registros inválidos:      {registros_invalidos}\n")
            relatorio.write(f"Classificação NORMAL:     {categorias['NORMAL']}\n")
            relatorio.write(f"Classificação ATENCAO:    {categorias['ATENCAO']}\n")
            relatorio.write(f"Classificação SUSPEITO:   {categorias['SUSPEITO']}\n")
            relatorio.write("==================================================\n")

        print(f"\n[SUCESSO] Relatório salvo com sucesso em '{caminho_saida}'.")
    except OSError as erro:
        print(f"\n[ERRO] Não foi possível salvar o relatório: {erro}")


def main():
    arquivo_entrada = "eventos.csv"
    try:
        eventos, invalidos = analisar_csv(arquivo_entrada)
        gerar_e_salvar_relatorio(eventos, invalidos)
    except FileNotFoundError:
        print(f"[ERRO] Arquivo '{arquivo_entrada}' não foi encontrado.")
    except ValueError as erro_val:
        print(f"[ERRO DE VALIDAÇÃO] {erro_val}")
    except OSError as erro:
        print(f"[ERRO DE I/O] Erro ao acessar arquivo: {erro}")


if __name__ == "__main__":
    main()
